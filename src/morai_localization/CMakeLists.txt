cmake_minimum_required(VERSION 3.0.2)
project(morai_localization)

add_compile_options(-Wall -Wextra -Wpedantic)

find_package(catkin REQUIRED COMPONENTS
  geometry_msgs
  morai_udp_bridge
  nav_msgs
  roscpp
  sensor_msgs
  std_msgs
  tf2_ros
)

catkin_package(
  INCLUDE_DIRS include
  LIBRARIES morai_utm_projector morai_imu_orientation morai_velocity_estimator
  CATKIN_DEPENDS geometry_msgs morai_udp_bridge nav_msgs roscpp sensor_msgs
                 std_msgs tf2_ros
)

include_directories(
  include
  ${catkin_INCLUDE_DIRS}
)

add_library(morai_utm_projector
  src/projection/utm_projector.cpp
)
target_compile_features(morai_utm_projector PUBLIC cxx_std_14)
target_link_libraries(morai_utm_projector
  ${catkin_LIBRARIES}
)

add_library(morai_imu_orientation
  src/orientation/imu_orientation.cpp
)
target_compile_features(morai_imu_orientation PUBLIC cxx_std_14)
target_link_libraries(morai_imu_orientation
  ${catkin_LIBRARIES}
)

add_library(morai_velocity_estimator
  src/estimation/velocity_estimator.cpp
)
target_compile_features(morai_velocity_estimator PUBLIC cxx_std_14)
target_link_libraries(morai_velocity_estimator
  ${catkin_LIBRARIES}
)

add_executable(gps_utm_projector_node
  src/nodes/gps_utm_projector_node.cpp
)
target_compile_features(gps_utm_projector_node PUBLIC cxx_std_14)
target_link_libraries(gps_utm_projector_node
  morai_utm_projector
  ${catkin_LIBRARIES}
)

add_executable(imu_orientation_adapter_node
  src/nodes/imu_orientation_adapter_node.cpp
)
target_compile_features(imu_orientation_adapter_node PUBLIC cxx_std_14)
target_link_libraries(imu_orientation_adapter_node
  morai_imu_orientation
  ${catkin_LIBRARIES}
)

add_executable(localization_fusion_node
  src/nodes/localization_fusion_node.cpp
)
target_compile_features(localization_fusion_node PUBLIC cxx_std_14)
target_link_libraries(localization_fusion_node
  morai_imu_orientation
  morai_velocity_estimator
  ${catkin_LIBRARIES}
)
add_dependencies(localization_fusion_node ${catkin_EXPORTED_TARGETS})

install(
  TARGETS
    morai_utm_projector
    morai_imu_orientation
    morai_velocity_estimator
    gps_utm_projector_node
    imu_orientation_adapter_node
    localization_fusion_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION}
)

install(
  DIRECTORY include/${PROJECT_NAME}/
  DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION}
)

install(
  DIRECTORY config launch
  DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}
)
